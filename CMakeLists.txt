cmake_minimum_required(VERSION 3.24)
# IDE metadata only: this target never links or runs an agent.
if(APPLE)
    set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Supported architecture")
    set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0 CACHE STRING "Supported OS floor")
endif()
project(harness_ide LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(CMAKE_C_EXTENSIONS YES)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(VEXSPOKE_SOURCE_DIR "" CACHE PATH "Local Vexspoke src; no downloads")
file(GLOB_RECURSE sources CONFIGURE_DEPENDS "src/*.c" "src/*.h")
add_library(harness_ide OBJECT EXCLUDE_FROM_ALL ${sources})
target_include_directories(harness_ide PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(VEXSPOKE_SOURCE_DIR)
    target_include_directories(harness_ide PRIVATE "${VEXSPOKE_SOURCE_DIR}")
endif()
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(harness_ide PRIVATE -Wall -Wextra -Werror)
    if(APPLE)
        target_compile_options(harness_ide PRIVATE -mcpu=apple-m1)
    endif()
endif()
message(STATUS "Harness: indexing-only C23 target; runtime builds belong to b.")
